```

# 重新导入
mysql -u zabbix -p'your_strong_password' zabbix < database/mysql/schema.sql
mysql -u zabbix -p'your_strong_password' zabbix < database/mysql/images.sql
mysql -u zabbix -p'your_strong_password' zabbix < database/mysql/data.sql
```
